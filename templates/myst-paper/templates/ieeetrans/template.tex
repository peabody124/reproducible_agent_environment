%%%%%%%%%%%%%%%  IEEE Conf Margin Fixes   %%%%%%%%%%%%%%%
% IEEEtran.cls uses different margins than ieeeconf.cls (the official template).
% These overrides match the official ieeeconf.cls margins:
%   - Side margins: 0.75in (instead of IEEEtran's ~0.68in)
%   - First page top: ~1in (instead of IEEEtran's ~0.75in)
%   - Text width: 7.0in (instead of IEEEtran's 43pc/7.17in)
%   - Column separation: 0.2in (instead of IEEEtran's 1pc/0.167in)

% 1. Fix text width to match ieeeconf.cls (7.0in instead of 43pc)
\setlength{\textwidth}{7.0in}

% 2. Re-center text with 0.75in side margins
%    LaTeX has 1in built-in offset, so -0.25in gives 0.75in effective margin
\setlength{\oddsidemargin}{-0.25in}
\setlength{\evensidemargin}{-0.25in}

% 3. Add extra top space on first page to match ieeeconf.cls (~1in top margin)
%    ieeeconf.cls uses \vskip0.25in before the title; IEEEtran uses \IEEEtitletopspaceextra
\def\IEEEtitletopspaceextra{0.25in}

% 4. Fix column separation to match ieeeconf.cls (0.2in instead of 1pc)
\setlength{\columnsep}{0.2in}

% 5. Fix textheight to match ieeeconf.cls (9.5in instead of 9.25in)
%    This gives ~0.75in bottom margin instead of ~1.0in
%    ieeeconf.cls: textheight = 9.5in for conferences
\setlength{\textheight}{9.5in}
